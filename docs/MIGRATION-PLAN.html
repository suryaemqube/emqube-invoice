8. API Calling Convention

Every API call follows (from http.service.ts):

POST {baseURL}api/{controller}/{action}
Content-Type: application/json
Body: JSON.stringify(value)  // often null, a primitive, or a small object

No Authorization header. Role/user context passed in POST body. The new ApiService should match this convention, not REST.

Invoice Status Derivation — Detailed

// Pure function — no API call, no state
export function deriveInvoiceStatus(inv: InvoiceListItem): InvoiceStatus {
  const isCancelled = inv.InvoiceTotal == 0 && inv.TotalVAT == 0;
  const isPaid = inv.PaymentRecdDateString != null
                 && inv.PaymentRecdDateString.trim() !== '';
  if (isCancelled) return 'Cancelled';
  if (isPaid)      return 'Paid';
  return 'Due';
}

Pure Functions — No API Call Required

These are entirely client-side and should live in a utils/ or helpers/ directory, not a service:

  • deriveInvoiceStatus(inv) → 'Cancelled' | 'Paid' | 'Due'
  • computeLineItemTotal(item) → number
  • computeInvoiceSubTotal(lineItems) → number
  • computeVatAmount(subTotal, vatPercentage) → number
  • computeGrandTotal(subTotal, vatAmount) → number
  • computeConvertedAmount(baseAmount, conversionRate) → number
  • computeDueDate(issueDate, paymentTermsDays) → Date
  • computePayDueDate(dueDate, gracePeriodDays) → Date

Service Pattern

@Injectable({ providedIn: 'root' })
export class InvoiceService {
  private api = inject(ApiService);

  getList(req: InvoiceListRequest): Observable<InvoiceListItem[]> {
    return this.api.post<InvoiceListItem[]>('Invoice', 'GetInvoiceList', req);
  }

  getKPI(req: InvoiceListRequest): Observable<InvoiceKPI> {
    return this.api.post<InvoiceKPI>('Invoice', 'GetInvoiceKPI', req);
  }

  getById(id: number): Observable<Invoice> {
    return this.api.post<Invoice>('Invoice', 'GetInvoiceById', { InvoiceId: id });
  }

  save(invoice: Invoice): Observable<{ InvoiceId: number }> {
    return this.api.post<{ InvoiceId: number }>('Invoice', 'SaveInvoice', invoice);
  }
}
